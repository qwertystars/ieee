# T51 Spike: PostgreSQL extension coexistence evidence
# Date: 2026-07-18
# Image: resqnet-db (built from Dockerfile.db)
# Base: postgres:16-bookworm

## Build

$ docker build -t resqnet-db -f Dockerfile.db .
# Exit code: 0
# Build time: ~70 seconds (cached after first build)

## Container startup

$ docker run --rm -d --name resqnet-db-test -e POSTGRES_PASSWORD=disaster resqnet-db
# Container started successfully

## Extension creation

$ docker exec resqnet-db-test psql -U disaster -d disaster -c \
    "CREATE EXTENSION IF NOT EXISTS postgis; \
     CREATE EXTENSION IF NOT EXISTS age; \

## Functional tests

### PostGIS
$ docker exec resqnet-db-test psql -U disaster -d disaster -c \
    "SELECT ST_AsText(ST_Point(-73.9857, 40.7484)) AS postgis_test;"

 POINT(-73.9857 40.7484)
(1 row)

### h3 (note: takes point type, not separate lat/lng args)

### TimescaleDB
$ docker exec resqnet-db-test psql -U disaster -d disaster -c \
    "CREATE TABLE sensor_readings (ts TIMESTAMPTZ NOT NULL, sensor_id TEXT NOT NULL, value DOUBLE PRECISION); \
